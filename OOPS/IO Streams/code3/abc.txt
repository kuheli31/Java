import java.io.*;//read and write in same filepublic class RWDemo{	public static void main(String[] args) throws Exception	{		/*FileReader fr = new FileReader("RWDemo.java"); // to read same .java file		FileWriter fw = new FileWriter("abc.txt");*/		BufferedReader br = new BufferedReader(new FileReader("RWDemo.java")); // to read same .java file		BufferedWriter bw = new BufferedWriter(new FileWriter("abc.txt"));		int lcount=0;		String line;		//EOF - readLine() -> null		while((line = br.readLine()) != null)		{			bw.write(line); // write(int)			lcount++;		}		System.out.println("\n No. of lines in the file: "+lcount);		System.out.println("\nFile read and successfully written to abc.txt file");				br.close();		bw.close();		}}
